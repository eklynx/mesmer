cmake_minimum_required(VERSION 3.20)
project(BrunnerDirect LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /WX /EHsc)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
find_package(Threads REQUIRED)
file(GLOB BRUNNER_SOURCES RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "*.hpp" "*.cpp" "*.html")
list(APPEND BRUNNER_SOURCES CMakeLists.txt)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${BRUNNER_SOURCES})
list(SORT BRUNNER_SOURCES)
set(BRUNNER_ID_INPUT "")
foreach(source IN LISTS BRUNNER_SOURCES)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" digest)
  string(APPEND BRUNNER_ID_INPUT "${source}:${digest};")
endforeach()
string(SHA256 BRUNNER_ID "${BRUNNER_ID_INPUT}")
add_compile_definitions(BRUNNER_BUILD_ID="${BRUNNER_ID}")
# The workbench page, embedded as bytes at configure time (no string-literal
# limits). The generated file changes only when the page does.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/workbench.html" WORKBENCH_HEX HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f])" "\\1\n" WORKBENCH_HEX "${WORKBENCH_HEX}")
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," WORKBENCH_BYTES "${WORKBENCH_HEX}")
set(WORKBENCH_PAGE "${CMAKE_CURRENT_BINARY_DIR}/generated/workbench_page.cpp")
file(WRITE "${WORKBENCH_PAGE}.in" "// Generated from workbench.html by CMakeLists.txt.\n#include <cstddef>\nnamespace brunner::workbench {\nextern const unsigned char page_bytes[];\nextern const std::size_t page_size;\nconst unsigned char page_bytes[]={\n${WORKBENCH_BYTES}0x00};\nconst std::size_t page_size=sizeof(page_bytes)-1;\n}\n")
configure_file("${WORKBENCH_PAGE}.in" "${WORKBENCH_PAGE}" COPYONLY)
set(WORKBENCH_SOURCES workbench_server.cpp "${WORKBENCH_PAGE}")
enable_testing()
add_executable(brunner_hud_test hud_tests.cpp)
target_link_libraries(brunner_hud_test PRIVATE Threads::Threads)
add_test(NAME brunner_hud COMMAND brunner_hud_test)
add_executable(brunner_core_test tests.cpp)
add_test(NAME brunner_core COMMAND brunner_core_test)
add_executable(brunner_monitor_test monitor_tests.cpp)
add_test(NAME brunner_monitor COMMAND brunner_monitor_test)
add_executable(brunner_yoke_test yoke_tests.cpp)
add_test(NAME brunner_yoke COMMAND brunner_yoke_test)
add_executable(brunner_readonly_test readonly_tests.cpp hid_select.cpp reader.cpp)
target_link_libraries(brunner_readonly_test PRIVATE Threads::Threads)
add_test(NAME brunner_readonly COMMAND brunner_readonly_test)
add_executable(brunner_bridge_test bridge_tests.cpp hid_select.cpp)
target_link_libraries(brunner_bridge_test PRIVATE Threads::Threads)
add_test(NAME brunner_bridge COMMAND brunner_bridge_test)
add_executable(brunner_panel_preview panel_preview.cpp)
add_executable(brunner_workbench_test workbench_tests.cpp ${WORKBENCH_SOURCES})
target_link_libraries(brunner_workbench_test PRIVATE Threads::Threads)
# The models' shared vectors, generated from elevator.html and aileron.html (the *_vectors.mjs scripts).
target_compile_definitions(brunner_workbench_test PRIVATE BRUNNER_ELEVATOR_VECTORS="${CMAKE_CURRENT_SOURCE_DIR}/elevator_vectors.json"
    BRUNNER_AILERON_VECTORS="${CMAKE_CURRENT_SOURCE_DIR}/aileron_vectors.json"
    BRUNNER_ELEVATOR_ALPHA_VECTORS="${CMAKE_CURRENT_SOURCE_DIR}/elevator_alpha_vectors.json")
add_test(NAME brunner_workbench COMMAND brunner_workbench_test)
# Dev harness: the real Workbench and server with a simulated aircraft.
add_executable(brunner_workbench_demo workbench_demo.cpp ${WORKBENCH_SOURCES})
target_link_libraries(brunner_workbench_demo PRIVATE Threads::Threads)
if(WIN32)
  target_link_libraries(brunner_workbench_test PRIVATE ws2_32 iphlpapi)
  target_link_libraries(brunner_workbench_demo PRIVATE ws2_32 iphlpapi)
endif()
set(XPLANE_SDK "" CACHE PATH "Unpacked official X-Plane SDK directory")
if(APPLE)
  find_library(IOKIT_FRAMEWORK IOKit REQUIRED)
  find_library(COREFOUNDATION_FRAMEWORK CoreFoundation REQUIRED)
endif()
if(WIN32)
  if(NOT EXISTS "${XPLANE_SDK}/CHeaders/XPLM/XPLMPlugin.h")
    message(FATAL_ERROR "Set XPLANE_SDK to the official SDK directory")
  endif()
  add_library(BrunnerDirect MODULE plugin.cpp reader.cpp hid_select.cpp hid_win.cpp ${WORKBENCH_SOURCES})
  target_compile_definitions(BrunnerDirect PRIVATE IBM=1 APL=0 LIN=0 XPLM200 XPLM210 XPLM300 XPLM301)
  target_include_directories(BrunnerDirect PRIVATE "${XPLANE_SDK}/CHeaders/XPLM")
  target_link_libraries(BrunnerDirect PRIVATE hid setupapi opengl32 gdi32 ws2_32 iphlpapi "${XPLANE_SDK}/Libraries/Win/XPLM_64.lib")
  add_executable(brunner_hardware_read_test smoke_win.cpp reader.cpp hid_select.cpp hid_win.cpp)
  target_link_libraries(brunner_hardware_read_test PRIVATE hid setupapi)
  set_target_properties(BrunnerDirect PROPERTIES PREFIX "" SUFFIX ".xpl")
  set_property(TARGET BrunnerDirect brunner_core_test brunner_monitor_test brunner_hud_test brunner_yoke_test brunner_panel_preview brunner_hardware_read_test brunner_workbench_test brunner_workbench_demo PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
elseif(APPLE AND XPLANE_SDK)
  # macOS plugin (build_macos.sh); the yoke is reached through hid_mac.cpp (IOKit).
  if(NOT EXISTS "${XPLANE_SDK}/CHeaders/XPLM/XPLMPlugin.h")
    message(FATAL_ERROR "XPLANE_SDK is not the official SDK directory: ${XPLANE_SDK}")
  endif()
  find_library(XPLM_FRAMEWORK XPLM PATHS "${XPLANE_SDK}/Libraries/Mac" NO_DEFAULT_PATH REQUIRED)
  find_library(OPENGL_FRAMEWORK OpenGL REQUIRED)
  add_library(BrunnerDirect MODULE plugin.cpp reader.cpp hid_select.cpp hid_mac.cpp ${WORKBENCH_SOURCES})
  target_compile_definitions(BrunnerDirect PRIVATE IBM=0 APL=1 LIN=0 XPLM200 XPLM210 XPLM300 XPLM301)
  target_include_directories(BrunnerDirect PRIVATE "${XPLANE_SDK}/CHeaders/XPLM")
  target_link_libraries(BrunnerDirect PRIVATE Threads::Threads "${XPLM_FRAMEWORK}" "${OPENGL_FRAMEWORK}" "${IOKIT_FRAMEWORK}" "${COREFOUNDATION_FRAMEWORK}")
  set_target_properties(BrunnerDirect PROPERTIES PREFIX "" SUFFIX ".xpl" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
endif()
if(APPLE)
  # Needs no X-Plane SDK: the macOS yoke reader and its read-only hardware check.
  add_executable(brunner_hardware_read_test smoke_mac.cpp reader.cpp hid_select.cpp hid_mac.cpp)
  target_link_libraries(brunner_hardware_read_test PRIVATE Threads::Threads "${IOKIT_FRAMEWORK}" "${COREFOUNDATION_FRAMEWORK}")
endif()
